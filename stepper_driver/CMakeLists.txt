cmake_minimum_required(VERSION 3.8)
project(stepper_driver)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_executable(stepper_driver src/stepper_driver.cpp)
ament_target_dependencies(stepper_driver rclcpp std_msgs trajectory_msgs)
install(TARGETS stepper_driver DESTINATION lib/${PROJECT_NAME})
ament_package()
