cmake_minimum_required(VERSION 3.8)
project(moveit_motion_controller)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_executable(moveit_motion_controller src/moveit_motion_controller.cpp)
ament_target_dependencies(moveit_motion_controller rclcpp std_msgs trajectory_msgs)
install(TARGETS moveit_motion_controller DESTINATION lib/${PROJECT_NAME})
ament_package()
